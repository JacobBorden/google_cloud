cmake_minimum_required(VERSION 3.14)
project(google_cloud LANGUAGES CXX)
include(CTest)
find_package(OpenSSL REQUIRED)
add_library(socket INTERFACE)
target_compile_features(socket INTERFACE cxx_std_20)
target_include_directories(socket INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(socket INTERFACE OpenSSL::SSL OpenSSL::Crypto)
add_executable(main.exe main.cpp)
target_link_libraries(main.exe PRIVATE socket)
if(BUILD_TESTING)
  add_executable(socket_tests tests/socket_tests.cpp)
  target_link_libraries(socket_tests PRIVATE socket)
  target_compile_options(socket_tests PRIVATE -Wall -Wextra)
  foreach(group request lifecycle loopback)
    add_test(NAME socket_${group} COMMAND socket_tests ${group})
    set_tests_properties(socket_${group} PROPERTIES TIMEOUT 15)
  endforeach()
endif()
